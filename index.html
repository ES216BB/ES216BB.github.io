// ------------------------------------------------------------------- //
/// 0. EquipmentLoan Class Definition
// ------------------------------------------------------------------- //
class EquipmentLoan {
private:
    string EquipmentName;
    string LaboratoryName;
    int MaxBorrowers;
    string* BorrowerList;
    int BorrowerCount;

public:
    EquipmentLoan();
    ~EquipmentLoan();
    void SetDetails(string EN, string LN, int MB);
    void AddBorrower(string BorrowerName);
    void DisplayEquipmentDetails();
    void DisplayBorrowerList();
    float LoanRevenue(float LoanFee);
};

// ------------------------------------------------------------------- //
/// 1. Constructor Function
// ------------------------------------------------------------------- //
EquipmentLoan::EquipmentLoan() {
    EquipmentName = "";
    LaboratoryName = "";
    MaxBorrowers = 0;
    BorrowerList = nullptr;
    BorrowerCount = 0;
}

// ------------------------------------------------------------------- //
/// 2. Destructor Function
// ------------------------------------------------------------------- //
EquipmentLoan::~EquipmentLoan() {
    if (BorrowerList != nullptr) {
        delete[] BorrowerList;
        BorrowerList = nullptr;
    }
}

// ------------------------------------------------------------------- //
/// 3. Set Details Function
// ------------------------------------------------------------------- //
void EquipmentLoan::SetDetails(string EN, string LN, int MB) {
    EquipmentName = EN;
    LaboratoryName = LN;
    MaxBorrowers = MB;
    BorrowerCount = 0;
    
    if (MB > 0) {
        BorrowerList = new string[MB];
    } else {
        BorrowerList = nullptr;
    }
}

// ------------------------------------------------------------------- //
/// 4. AddBorrower Function
// ------------------------------------------------------------------- //
void EquipmentLoan::AddBorrower(string BorrowerName) {
    if (BorrowerCount < MaxBorrowers) {
        BorrowerList[BorrowerCount] = BorrowerName;
        BorrowerCount++;
    } else {
        cout << "Maximum borrowers reached: " << BorrowerName << " not added." << endl;
    }
}

// ------------------------------------------------------------------- //
/// 5. DisplayEquipmentDetails Function
// ------------------------------------------------------------------- //
void EquipmentLoan::DisplayEquipmentDetails() {
    cout << "Equipment Name: " << EquipmentName << endl;
    cout << "Laboratory: " << LaboratoryName << endl;
    cout << "Max Borrowers: " << MaxBorrowers << endl;
    cout << "Current Borrowers: " << BorrowerCount << endl;
}

// ------------------------------------------------------------------- //
/// 6. DisplayBorrowerList Function
// ------------------------------------------------------------------- //
void EquipmentLoan::DisplayBorrowerList() {
    if (BorrowerCount == 0) {
        cout << "No borrowers registered." << endl;
    } else {
        cout << "Borrower List: ";
        for (int i = 0; i < BorrowerCount; i++) {
            cout << BorrowerList[i] << " ; ";
        }
        cout << endl;
    }
}

// ------------------------------------------------------------------- //
/// 7. LoanRevenue Function
// ------------------------------------------------------------------- //
float EquipmentLoan::LoanRevenue(float LoanFee) {
    return BorrowerCount * LoanFee;
}
