TGeyserEnergy Management System C++ Source Code

#include <iostream>
#include <fstream>
#include <string>
#include <iomanip>

using namespace std;

// Structure Definition
struct TGeyserEnergy {
    int geyserID;
    float ratedPower;
    float energyUsed;
    TGeyserEnergy* next;

    // Member function declaration
    void InitialiseGeyser(int id, float power, float energy);
};

// 1. InitialiseGeyser Function
void TGeyserEnergy::InitialiseGeyser(int id, float power, float energy) {
    geyserID = id;
    ratedPower = power;
    energyUsed = energy;
    next = nullptr;
}

// 2. AddGeyserToList Function
TGeyserEnergy* AddGeyserToList(TGeyserEnergy **startNode) {
    TGeyserEnergy* newNode = new TGeyserEnergy();
    
    if (*startNode == nullptr) {
        *startNode = newNode;
    } else {
        TGeyserEnergy* temp = *startNode;
        while (temp->next != nullptr) {
            temp = temp->next;
        }
        temp->next = newNode;
    }
    return newNode;
}

// 7. DeleteAllGeysers Function
void DeleteAllGeysers(TGeyserEnergy **startNode) {
    TGeyserEnergy* current = *startNode;
    while (current != nullptr) {
        TGeyserEnergy* nextNode = current->next;
        delete current;
        current = nextNode;
    }
    *startNode = nullptr;
}

// 3. ReadDataFromFile Function
void ReadDataFromFile(string filename, TGeyserEnergy **startNode) {
    DeleteAllGeysers(startNode);
    
    ifstream inFile(filename);
    if (!inFile) {
        cout << "Error opening file: " << filename << endl;
        return;
    }

    int id;
    float power, energy;
    
    while (inFile >> id >> power >> energy) {
        TGeyserEnergy* newNode = AddGeyserToList(startNode);
        newNode->InitialiseGeyser(id, power, energy);
    }
    
    inFile.close();
}

// 4. DisplayGeyserData Function
void DisplayGeyserData(TGeyserEnergy *startNode) {
    if (startNode == nullptr) {
        cout << "No data loaded...." << endl;
        return;
    }

    cout << left << setw(15) << "Geyser ID" 
         << setw(20) << "Rated power (kW)" 
         << setw(25) << "Energy consumption (kWh)" << endl;

    TGeyserEnergy* temp = startNode;
    while (temp != nullptr) {
        cout << left << setw(15) << temp->geyserID 
             << setw(20) << temp->ratedPower 
             << setw(25) << temp->energyUsed << endl;
        temp = temp->next;
    }
}

// 5. AverageEnergyConsumption Function
float AverageEnergyConsumption(TGeyserEnergy *startNode) {
    if (startNode == nullptr) {
        return 0.0f;
    }

    float totalEnergy = 0.0f;
    int count = 0;
    TGeyserEnergy* temp = startNode;

    while (temp != nullptr) {
        totalEnergy += temp->energyUsed;
        count++;
        temp = temp->next;
    }

    return totalEnergy / count;
}

// 6. DeleteSingleGeyser Function
void DeleteSingleGeyser(TGeyserEnergy **startNode) {
    if (*startNode == nullptr) {
        return;
    }
    
    TGeyserEnergy* temp = *startNode;
    *startNode = (*startNode)->next;
    delete temp;
}
