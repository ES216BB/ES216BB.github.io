// ------------------------------------------------------------------- //
/// 0. WorkshopRegistration Class Definition
// ------------------------------------------------------------------- //
class WorkshopRegistration {
private:
    string WorkshopTitle;
    string PresenterName;
    int MaxParticipants;
    string* ParticipantList;
    int ParticipantCount;

public:
    WorkshopRegistration();
    ~WorkshopRegistration();
    void SetDetails(string WT, string PN, int MP);
    void AddParticipant(string ParticipantName);
    void DisplayWorkshopDetails();
    void DisplayParticipantList();
    float RegistrationRevenue(float RegistrationFee);
};

// ------------------------------------------------------------------- //
/// 1. Constructor Function
// ------------------------------------------------------------------- //
WorkshopRegistration::WorkshopRegistration() {
    WorkshopTitle = "";
    PresenterName = "";
    MaxParticipants = 0;
    ParticipantList = nullptr;
    ParticipantCount = 0;
}

// ------------------------------------------------------------------- //
/// 2. Destructor Function
// ------------------------------------------------------------------- //
WorkshopRegistration::~WorkshopRegistration() {
    delete[] ParticipantList;
}

// ------------------------------------------------------------------- //
/// 3. Set Details Function
// ------------------------------------------------------------------- //
void WorkshopRegistration::SetDetails(string WT, string PN, int MP) {
    WorkshopTitle = WT;
    PresenterName = PN;
    MaxParticipants = MP;
    ParticipantList = new string[MaxParticipants];
}

// ------------------------------------------------------------------- //
/// 4. AddParticipant Function
// ------------------------------------------------------------------- //
void WorkshopRegistration::AddParticipant(string ParticipantName) {
    if (ParticipantCount < MaxParticipants) {
        ParticipantList[ParticipantCount] = ParticipantName;
        ParticipantCount++;
    } else {
        cout << "Maximum participants reached: " << ParticipantName << " not added." << endl;
    }
}

// ------------------------------------------------------------------- //
/// 5. DisplayWorkshopDetails Function
// ------------------------------------------------------------------- //
void WorkshopRegistration::DisplayWorkshopDetails() {
    cout << "Workshop Title: " << WorkshopTitle << endl;
    cout << "Presenter: " << PresenterName << endl;
    cout << "Max Participants: " << MaxParticipants << endl;
    cout << "Current Participants: " << ParticipantCount << endl;
}

// ------------------------------------------------------------------- //
/// 6. DisplayParticipantList Function
// ------------------------------------------------------------------- //
void WorkshopRegistration::DisplayParticipantList() {
    if (ParticipantCount == 0) {
        cout << "No participants registered." << endl;
    } else {
        cout << "Participant List: ";
        for (int i = 0; i < ParticipantCount; i++) {
            cout << ParticipantList[i] << " ; ";
        }
        cout << endl;
    }
}

// ------------------------------------------------------------------- //
/// 7. RegistrationRevenue Function
// ------------------------------------------------------------------- //
float WorkshopRegistration::RegistrationRevenue(float RegistrationFee) {
    return ParticipantCount * RegistrationFee;
}
